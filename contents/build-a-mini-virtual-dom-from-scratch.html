Build a Mini Virtual DOM from Scratch

Let's start building.

Our first goal is to create a function that takes a string that looks like a CSS selector and returns an object. The object shape is simple: { tag, attrs, children }. Extra attributes can be passed as a second argument and merged in. Children (nested nodes or text) come last.

Here is the function implemented:

// A text scanner that reads a shorthand description like
// "div#app.container[type=text]" and splits it into recognizable pieces —
// the tag name, the ID, class names, and bracket-style attributes.
// It walks through the string chunk by chunk, one piece at a time.
const selParser = /(?:(^|#|\.)([^#\.\[\]]+))|(\[.+?\])/g;

// A secondary scanner that zooms into just the bracket part
// of a shorthand — e.g. [type=text] or [checked] — and pulls out the
// attribute name and its value (if any). Used after selParser flags a bracket.
const attrParser = /\[(.+?)(?:=("|'|)(.+?)\2)?\]/;

// We borrow toString from a plain object so we can reliably identify the true
// type of any value passed in. typeof is unreliable — arrays, null, and plain
// objects all return "object". {}.toString.call(x) returns a precise tag like
// "[object Object]" or "[object Array]" that can't be faked.
const toString = {}.toString;

const createVn = function () {
  const a = arguments;

  // The function signature is createVn(selector, attrs?, children?). When attrs is
  // omitted, the second argument is children. We check here which case we're
  // in by seeing if a[1] is a plain object. toString.call is the reliable way
  // to do this — it won't confuse arrays, vnodes, or strings for an attrs map.
  const hasAttrs = toString.call(a[1]) == "[object Object]";
  const attrs = hasAttrs ? a[1] : {};

  // React uses "className"; raw DOM and some callers use "class". We detect
  // which convention the caller is using so we can merge classes consistently
  // throughout — if they pass { className: "foo" } we never write to "class",
  // and vice versa.
  const clsKey = "class" in attrs ? "class" : "className";

  let m;
  let classes = [];

  const vn = { tag: "div", attrs: {} };

  // selParser walks through the selector string one chunk at a time.
  // Each match gives us a prefix in m[1] and a name in m[2], or a bracket
  // attribute in m[3]:
  //
  //   m[1] == ""   → the very first token, which is the tag name  ("div")
  //   m[1] == "#"  → an ID fragment                               (#app)
  //   m[1] == "."  → a class fragment                             (.container)
  //   m[3]         → a bracket attribute                           ([type=text])
  //
  // Classes are collected into an array and joined at the end because a single
  // selector can have multiple: "div.foo.bar" → "foo bar".
  // Bracket attributes are handed to attrParser for a second-level parse;
  // a bare flag like [checked] gets the value true (no string value present).
  while ((m = selParser.exec(a[0]))) {
    if (m[1] == "") vn.tag = m[2];
    else if (m[1] == "#") vn.attrs.id = m[2];
    else if (m[1] == ".") classes.push(m[2]);
    else if (m[3][0] == "[") {
      var p = attrParser.exec(m[3]);
      vn.attrs[p[1]] = p[3] || true;
    }
  }

  if (classes.length) vn.attrs["className"] = classes.join(" ");

  vn.children = hasAttrs ? a[2] : a[1];

  for (var k in attrs) {
    if (k == clsKey) vn.attrs[k] = (vn.attrs[k] || "") + " " + attrs[k];
    else vn.attrs[k] = attrs[k];
  }

  return vn;
};

Now let's run the code above with a couple of test cases and see if it works.

  • createVn("div") should return { tag: "div", attrs: {}, children: undefined }
  • createVn("div#app.container") should return { tag: "div", attrs: { id: "app", class: "container" }, children: undefined }
  • createVn("input[type=text]", { placeholder: "Name" }) should return { tag: "input", attrs: { type: "text", placeholder: "Name" }, children: undefined }
  • createVn("ul", [createVn("li", "a"), createVn("li", "b")]) should return a ul vnode with two li children.

Now that we have verified the code works as expected, let's alias createVn to cvn for brevity.

Before we move to the next step, there are two constraints we must add:

  1. Parsing the selector string is expensive — the same selector can be called thousands of times per render, so we should cache the parse result.
  2. The cached result must never be mutated directly, since callers will add things like id, class, and children — always return a clone.

Here is the updated code:

const cache = {};

// A shallow copy helper — copies all keys from one object into a new one.
// We use this to clone the cached vnode so the original is never touched.
function clone(o) {
  const r = {};
  for (const k in o) r[k] = o[k];
  return r;
}

const createVn = function () {
  const a = arguments;
  const hasAttrs = toString.call(a[1]) == "[object Object]";
  const attrs = hasAttrs ? a[1] : {};
  const clsKey = "class" in attrs ? "class" : "className";

  // Check the cache first. If we've already parsed this selector string,
  // skip the parsing loop entirely.
  let vn = cache[a[0]];

  if (!vn) {
    // Not cached yet — parse and store the result.
    cache[a[0]] = vn = { tag: "div", attrs: {} };
    let m;
    const classes = [];

    while ((m = selParser.exec(a[0]))) {
      if (m[1] == "") vn.tag = m[2];
      else if (m[1] == "#") vn.attrs.id = m[2];
      else if (m[1] == ".") classes.push(m[2]);
      else if (m[3][0] == "[") {
        var p = attrParser.exec(m[3]);
        vn.attrs[p[1]] = p[3] || true;
      }
    }

    if (classes.length) vn.attrs[clsKey] = classes.join(" ");
  }

  // Clone the cached result so we never mutate what's stored in cache.
  vn = clone(vn);
  vn.attrs = clone(vn.attrs);

  vn.children = hasAttrs ? a[2] : a[1];

  for (var k in attrs) {
    if (k == clsKey) vn.attrs[k] = (vn.attrs[k] || "") + " " + attrs[k];
    else vn.attrs[k] = attrs[k];
  }

  return vn;
};

Here is the data flow sketch of the code above:

  "div#app.container"
          |
    [parse once, cache]
          |
    { tag: "div", attrs: { id: "app", class: "container" } }
          |
    [clone it]
          |
    [merge in caller's attrs]
          |
    [attach children]
          |
    return vnode

So the key thing we are doing is parse once and clone always.

Now we have a vnode factory. Nothing turns those objects into real DOM yet. No renderer. No attribute writing. No text nodes.

Our next problem is to create something that walks the vnode tree and calls real browser APIs to build actual DOM from it.

We have to build three new pieces into the code above:

  1. renderVNode(parent, vnode, index) — the recursive workhorse. Given a parent DOM node and a vnode, it creates a DOM element for object vnodes, a text node for string or number vnodes, handles array vnodes by iterating and recursing each item, and inserts the created node at the right position via insertBefore.
  2. setAttrs(node, attrs) — applies vnode.attrs onto a real DOM node. Three cases: a style key whose value is an object gets iterated rule by rule; a key that exists on the node is assigned as a DOM property (covering id, className, onclick, etc.); anything else falls back to setAttribute.
  3. render(root, vnode) — the public entry point. Clears root completely (no diffing yet), then calls renderVNode to build and insert the full tree.

Here are the functions implemented:

// Takes a real DOM node and a plain attrs object like { id: "app", onclick: fn, style: { color: "red" } }
// and applies every key onto the node. Three different cases depending on the key:
function setAttrs(node, attrs) {
  for (const k in attrs) {
    const v = attrs[k];
    if (k == "style") {
      // style is special — its value is itself an object like { color: "red", fontSize: "14px" }.
      // We have to loop inside it and set each rule one at a time on node.style.
      for (const rule in v) node.style[rule] = v[rule];
    } else if (k in node) {
      // If the key already exists on the DOM node (or anywhere up its prototype chain),
      // set it as a property directly. This covers id, className, value, checked,
      // and all event handlers like onclick, oninput — they live on the prototype.
      node[k] = v;
    } else {
      // Key is not a known DOM property — fall back to setAttribute.
      // This handles custom HTML attributes like data-* or aria-*.
      node.setAttribute(k, v);
    }
  }
}

// Turns a single vnode (or array of vnodes, or a plain string) into real DOM
// and appends it to parent at position index.
// It calls itself recursively to handle nested children — the vnode tree
// mirrors the DOM tree, so we just walk it depth-first.
function renderVNode(parent, vnode, index) {
  if (vnode == null) return; // nothing to render

  const dt = toString.call(vnode);

  if (dt == "[object Array]") {
    // A list of siblings — render each one in order.
    // i becomes the index so each child lands in the right slot.
    for (let i = 0; i < vnode.length; i++) renderVNode(parent, vnode[i], i);
  } else if (dt == "[object Object]") {
    // A vnode object like { tag: "div", attrs: {}, children: ... }
    // — create a real DOM element from it.
    if (typeof vnode.tag != "string") return; // safety guard: skip malformed vnodes

    const node = document.createElement(vnode.tag); // e.g. creates <div>

    setAttrs(node, vnode.attrs); // stamp id, class, events etc. onto it

    if (vnode.children != null) renderVNode(node, vnode.children, 0); // recurse: build children inside this node

    // Place the finished node into the parent at the right position.
    // childNodes[index] is the existing node we want to insert before;
    // if it doesn't exist (we're at the end) insertBefore treats null as "append".
    parent.insertBefore(node, parent.childNodes[index] || null);
  } else {
    // Anything that isn't an array or object is treated as text — a string like "hello"
    // or a number. Wrap it in a text node so the browser can display it.
    const node = document.createTextNode(vnode);
    parent.insertBefore(node, parent.childNodes[index] || null);
  }
}

// Public entry point. Call this with a real DOM container and a vnode tree
// to stamp the entire UI into the page.
function render(root, vnode) {
  root.innerHTML = ""; // wipe whatever was there before (no diffing yet)
  renderVNode(root, vnode, 0);
}

If we run the code below, we can see the form rendered using nothing but the script above.

Now let's look at the updated data flow sketch:

  render(root, vnode)
          |
    root.innerHTML = ""                        ← wipe (no diffing)
          |
    renderVNode(root, vnode, 0)
          |
    what type is vnode?
      ├── Array    → iterate → renderVNode(parent, child, i)    [recurse each]
      ├── Object   → cvn("div#app.container")
      │                  |
      │               [parse once, cache]
      │                  |
      │               { tag: "div", attrs: { id: "app", class: "container" } }
      │                  |
      │               [clone it]
      │                  |
      │               [merge in caller's attrs]
      │                  |
      │               [attach children]
      │                  |
      │             createElement(vnode.tag)
      │                  |
      │             setAttrs(node, vnode.attrs)
      │                  |
      │             renderVNode(node, vnode.children, 0)         [recurse]
      │                  |
      │             parent.insertBefore(node, childNodes[index])
      └── string / number → createTextNode(vnode)
                         |
                    parent.insertBefore(node, childNodes[index])

We now have two things: a way to describe UI as plain JavaScript objects, and a renderer that turns those objects into real DOM.
